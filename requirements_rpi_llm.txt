# GlaDOS — Raspberry Pi with cloud LLM (Anthropic/OpenAI)
#
# Everything in requirements_rpi.txt plus LLM client libraries and memory.
# No local model needed — connects to Anthropic Claude, OpenAI, or any
# OpenAI-compatible API (e.g. remote LM Studio on another machine).
#
# Install: pip install -r requirements_rpi_llm.txt
# Run:     python main.py
#
# Config example (Anthropic Claude):
#   client_type: ANTHROPIC
#   model: "claude-sonnet-4-20250514"
#   # Set ANTHROPIC_API_KEY env var
#
# Config example (remote LM Studio on another machine):
#   client_type: OPENAI
#   completion_url: "http://192.168.1.100:1234/v1"
#   model: "qwen/qwen3-30b-a3b-2507"

# ── Base RPi requirements ─────────────────────────
-r requirements_rpi.txt

# ── LLM Clients ──────────────────────────────────
openai>=1.59.8                 # OpenAI-compatible API client (LM Studio, Ollama, OpenAI)
anthropic>=0.43.0              # Anthropic Claude API client

# ── Memory System ─────────────────────────────────
chromadb>=0.4.0                # Persistent vector memory (remember/recall)
sentence-transformers>=2.2.0   # Embedding model for memory + knowledge RAG (downloads ~90MB model)

# ── Knowledge RAG (optional) ─────────────────────
# Uncomment if you want Qdrant-backed knowledge retrieval:
# qdrant-client>=1.7.0
# beautifulsoup4>=4.12.0       # For ZIM ingestion
# libzim>=3.0.0                # For ZIM ingestion

# ── NOT included ──────────────────────────────────
# torch                        # Not needed for RPi (Whisper uses tiny/base model)
# langchain, langgraph          # Only if using LangChain backend
# ollama                        # Only if using Ollama native API (use openai client instead)
# mistralai                     # Only if using Mistral native API
